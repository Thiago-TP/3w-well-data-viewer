\newcommand{\isquestion}{\wspage{Instances page} \raisebox{.75pt}{$\blacktriangleright$}~Is fault behavior consistent across instances and unique to the fault?}

% A fault class's dot, in its color on the viewer's pages, and its entry in
% the legend; class 10 is a well's first instance, which has no past to name
% it.
\newcommand{\isdot}[1]{\tikz[baseline=-.6ex]\fill[isc#1] (0,0) circle (1.7pt);}
\newcommand{\isentry}[2]{\isdot{#1}\,#1~#2}
\colorlet{isc10}{ws3wgrey}

% A map: its box (width and height, in the proportions scripts/instances.py
% scales the maps into, 1.68 by 1.08 with the margins), no axes, a thin
% frame; and the dots of one class, in its color.
\pgfplotsset{
  is map/.style 2 args={
    scale only axis, hide axis, width=#1, height=#2, anchor=south west,
    xmin=-0.04, xmax=1.64, ymin=-0.04, ymax=1.04, clip=false,
    axis background/.style={draw=ws3wgrey!80, line width=.3pt}},
  is class/.style 2 args={
    only marks, mark=*, mark size=#2,
    mark options={draw=white, line width=.05pt, fill=isc#1}},
}
% \isplot[dot size]{map}{column}: the dots of one map (figures/instances/maps/
% <map>.csv), colored by a column (the fault, or the class the well's past
% names), class by class in the order \isClasses<column> lists them. A class
% is a plot of its own, filtered out of the map's table: its dots then make
% one path, where a scatter plot of 1,119 dots each styled on its own outgrows
% pdfLaTeX's memory. (Expanded first, so that a map named through \ismode or
% \isBestKey reaches the file name as its name.)
\newcommand{\isplot}[3][.5pt]{%
  \edef\isnext{\noexpand\pgfplotsinvokeforeach{\csname isClasses#3\endcsname}{%
    \noexpand\addplot [is class={####1}{#1},
                       restrict expr to domain={\noexpand\thisrow{#3}}{####1:####1}]
      table [x=x, y=y] {figures/instances/maps/#2.csv};}}%
  \isnext}
