%% Appendix: what the sections count but have no time to show.
%%
%% Timelines: the label changes across an unlabeled stretch.
%% The Timelines section counts the unlabeled stretches a joined history keeps
%% between two different labels (\tlGapChangeCount on 3W 2.0.0) and the returns
%% to normal among them; here they are listed, then shown one by one in the
%% instance window, around the change. The table, the crops and their captions
%% come from tables/appendix/changes.csv and figures/appendix/, written by
%% scripts/appendix.py.
%%
%% Availability: every variable's samples and instances, live, frozen and
%% missing, as counts and shares, from figures/availability/coverage.csv.

\pgfplotstableread{tables/appendix/changes.csv}\apchanges

% \apchange{n}: the crop of change n (from 1) over its caption, half the
% width of the frame; the caption is read off the table.
\NewDocumentCommand{\apchange}{m}{%
  \pgfmathtruncatemacro{\aprow}{#1-1}%
  \pgfplotstablegetelem{\aprow}{label}\of\apchanges\let\aplabel\pgfplotsretval
  \pgfplotstablegetelem{\aprow}{at}\of\apchanges\let\apat\pgfplotsretval
  \pgfplotstablegetelem{\aprow}{before}\of\apchanges\let\apbefore\pgfplotsretval
  \pgfplotstablegetelem{\aprow}{after}\of\apchanges\let\apafter\pgfplotsretval
  \pgfplotstablegetelem{\aprow}{unlabeled}\of\apchanges\let\apunlabeled\pgfplotsretval
  \begin{minipage}[t]{0.49\linewidth}
    \includegraphics[width=\linewidth]{figures/appendix/change-#1.png}\par
    \vskip .5mm
    {\tiny\raggedright\hyphenpenalty=10000
     \textbf{#1}\quad \aplabel, \apat: \apbefore{} $\rightarrow$ \apunlabeled{} unlabeled
     $\rightarrow$ \apafter\par}
  \end{minipage}%
}

% \apgrid{a}{b}{c}{d}: four changes, two by two.
\NewDocumentCommand{\apgrid}{m m m m}{%
  \apchange{#1}\hfill\apchange{#2}\par
  \vskip 3mm
  \apchange{#3}\hfill\apchange{#4}%
}
